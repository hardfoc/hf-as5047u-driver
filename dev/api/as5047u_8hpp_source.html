157 static constexpr uint8_t ComputeCRC8(uint16_t data16) {
158 uint8_t crc = 0xC4;
159 for (int i = 0; i < 16; ++i) {
160 bool bit = ((data16 >> (15 - i)) & 1) ^ ((crc >> 7) & 1);
161 crc = (crc << 1) ^ (bit ? 0x1D : 0x00);
162 }
163 return crc ^ 0xFF;
164 }
